# Compile-only integration project: exercise the exact production build policy
# without linking the engine, loading a model, or requiring an accelerator.
cmake_minimum_required(VERSION 3.25)
project(GpuSpillGuardProbe LANGUAGES ${PROBE_LANGUAGE})
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(PROBE_CCACHE ccache)
if(PROBE_CCACHE)
    set(CMAKE_${PROBE_LANGUAGE}_COMPILER_LAUNCHER "${PROBE_CCACHE}")
endif()
set(CMAKE_${PROBE_LANGUAGE}_FLAGS_INTEGRATION "-O3 -g")
include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/GpuSpillGuard.cmake")
if(PROBE_LANGUAGE STREQUAL "CUDA")
    set(PROBE_SOURCE gpu_spill_probe.cu)
else()
    set(PROBE_SOURCE gpu_spill_probe.hip)
endif()
foreach(PROBE_VARIANT clean spill private helper scalar mixed clean_helper)
    add_library(probe_${PROBE_VARIANT} OBJECT ${PROBE_SOURCE})
endforeach()
target_compile_definitions(probe_spill PRIVATE PROBE_KIND=1)
target_compile_definitions(probe_private PRIVATE PROBE_KIND=2)
target_compile_definitions(probe_helper PRIVATE PROBE_KIND=3)
target_compile_definitions(probe_scalar PRIVATE PROBE_KIND=4)
target_compile_definitions(probe_mixed PRIVATE PROBE_KIND=5)
target_compile_definitions(probe_clean_helper PRIVATE PROBE_KIND=6)
